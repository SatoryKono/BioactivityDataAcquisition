name: bioetl-monitoring

# OPTIONAL operator stack (ADR-010). Not part of the default release bundle.
# Default local workflow: Python/venv only, or main stack alone.
# Removed from this surface: Loki, Promtail, Tempo, Quarantine Explorer.
#
# Local resource budget (32 GiB Windows hosts):
#   prometheus 3g + grafana 2g + renderer 3g + pushgateway 512m ≈ 8.5g cgroup caps
#   (was ~10g + 7.5g + 15g + 2.5g ≈ 35g — host thrash / OOM).

services:
  # Cap cgroup + TSDB retention so rule/dashboard peaks stay in-budget
  # (was mem_limit 10g with 8–10g peaks under recording-rule load).
  prometheus:
    image: prom/prometheus:v3.13.1@sha256:3c42b892cf723fa54d2f262c37a0e1f80aa8c8ddb1da7b9b0df9455a35a7f893
    container_name: bioetl-prometheus
    ports:
      - "127.0.0.1:9090:9090"
    environment:
      # Soft Go heap cap under mem_limit 3g (was unbounded → 8–10g peaks / host thrash).
      - GOMEMLIMIT=${PROMETHEUS_GOMEMLIMIT:-2500MiB}
      - GOGC=${PROMETHEUS_GOGC:-50}
    volumes:
      - ./grafana/prometheus.yml:/etc/prometheus/prometheus.yml:ro
      - ./grafana/prometheus-rules:/etc/prometheus/rules:ro
      - prometheus-data:/prometheus
    command:
      - "--config.file=/etc/prometheus/prometheus.yml"
      - "--storage.tsdb.path=/prometheus"
      - "--web.enable-admin-api"
      - "--storage.tsdb.retention.time=7d"
      - "--storage.tsdb.retention.size=2GB"
      # Local Desktop: fewer concurrent queries under rule load (was 4).
      - "--query.max-concurrency=2"
      - "--query.timeout=30s"
      - "--query.max-samples=5000000"
    healthcheck:
      test:
        [
          "CMD",
          "wget",
          "--quiet",
          "--tries=1",
          "--spider",
          "http://localhost:9090/-/healthy",
        ]
      interval: 30s
      timeout: 15s
      retries: 10
      start_period: 60s
    networks:
      - monitoring
    extra_hosts:
      - "host.docker.internal:host-gateway"
    restart: unless-stopped
    # Prefer killing renderer first (oom_score_adj 800); Prom is more valuable.
    oom_score_adj: 200
    mem_limit: 3g
    memswap_limit: 3g
    mem_reservation: 768m
    cpus: 2.0
    pids_limit: 2560
    init: true
    stop_grace_period: 120s
    logging:
      { driver: json-file, options: { max-size: "200m", max-file: "10" } }

  pushgateway:
    image: prom/pushgateway:v1.11.3@sha256:74fa117cef2d7e383112d25139ff1c2d2e309c35389a9e0554a47136a1482e48
    container_name: bioetl-pushgateway
    ports:
      - "127.0.0.1:9091:9091"
    networks:
      - monitoring
    restart: unless-stopped
    healthcheck:
      test:
        [
          "CMD",
          "wget",
          "--quiet",
          "--tries=1",
          "--spider",
          "http://localhost:9091/-/healthy",
        ]
      interval: 30s
      timeout: 15s
      retries: 10
      start_period: 60s
    mem_limit: 512m
    mem_reservation: 64m
    cpus: 0.5
    pids_limit: 512
    init: true
    stop_grace_period: 120s
    logging:
      { driver: json-file, options: { max-size: "100m", max-file: "10" } }

  grafana:
    image: grafana/grafana:12.0.0@sha256:263cbefd5d9b179893c47c415daab4da5c1f3d6770154741eca4f45c81119884
    container_name: bioetl-grafana
    ports:
      - "127.0.0.1:3000:3000"
    environment:
      - GF_SECURITY_ADMIN_PASSWORD=${GF_SECURITY_ADMIN_PASSWORD:?GF_SECURITY_ADMIN_PASSWORD is required}
      - GF_USERS_ALLOW_SIGN_UP=false
      - GF_METRICS_ENABLED=true
      - GF_LOG_LEVEL=info
      - GF_SERVER_ROOT_URL=http://localhost:3000
      - GF_SERVER_SERVE_FROM_SUB_PATH=false
      # Operator-facing clock on all shipped dashboards: YYYY-MM-DD HH:mm
      # (Grafana uses mm for minutes). Applies to time picker, axis ticks,
      # and default Time fields. Panel units use the same custom format.
      - GF_DATE_FORMATS_FULL_DATE=YYYY-MM-DD HH:mm
      - GF_DATE_FORMATS_INTERVAL_SECOND=HH:mm:ss
      - GF_DATE_FORMATS_INTERVAL_MINUTE=YYYY-MM-DD HH:mm
      - GF_DATE_FORMATS_INTERVAL_HOUR=YYYY-MM-DD HH:mm
      - GF_DATE_FORMATS_INTERVAL_DAY=YYYY-MM-DD
      - GF_DATE_FORMATS_INTERVAL_MONTH=YYYY-MM
      - GF_DATE_FORMATS_INTERVAL_YEAR=YYYY
      - GF_DATE_FORMATS_USE_BROWSER_LOCALE=false
      # Required for BioETL Ops HTTP panels (Run Explorer, identity tables, etc.).
      # Keep the frontend compatible with pinned Grafana 12.0.0. Infinity 3.11.3
      # imports react/jsx-runtime, which this Grafana build does not expose and
      # leaves the datasource-backed panels blank. Bootstrap enforces this pin.
      - BIOETL_INFINITY_PLUGIN_VERSION=${BIOETL_INFINITY_PLUGIN_VERSION:-3.8.0}
      # Remote renderer is optional for UI. When renderer is down/OOM, Grafana must
      # fail screenshot/PDF calls quickly — not hang worker threads (was unbounded
      # wait against dead Chromium). Dashboards keep working without renderer.
      - GF_RENDERING_SERVER_URL=http://renderer:8081/render
      - GF_RENDERING_CALLBACK_URL=http://grafana:3000/
      - GF_RENDERING_RENDERER_TOKEN=${GF_RENDERING_RENDERER_TOK
