## Machine-Checkable Coverage Gate

Required threshold per weak file: `>=85.0`.

```bash
./.venv/Scripts/python.exe -m coverage json -o .tmp_rf012_cov.json
./.venv/Scripts/python.exe - <<'PY'
import json
import sys

targets = {
    "src/bioetl/application/composite/conflict_resolver.py": 85.0,
    "src/bioetl/application/pipelines/uniprot/extractors/_comment_helpers.py": 85.0,
    "src/bioetl/application/services/dq_report_generation_mixin.py": 85.0,
    "src/bioetl/application/services/metadata_assemblers_helpers.py": 85.0,
}

data = json.load(open(".tmp_rf012_cov.json", encoding="utf-8"))["files"]
normalized = {
    key.replace("\\\\", "/"): value["summary"]["percent_covered"]
    for key, value in data.items()
}

errors = []
for path, threshold in targets.items():
    hit = next((p for p in normalized if p.endswith(path)), None)
    if hit is None:
        errors.append(f"{path}: missing from coverage report")
        continue
    pct = float(normalized[hit])
    if pct < threshold:
        errors.append(f"{path}: {pct:.2f}% < {threshold:.2f}%")

if errors:
    print("RF-012 coverage gate failed:")
    for error in errors:
        print(f"  - {error}")
    sys.exit(1)

print("RF-012 coverage gate: OK")
PY
```
