def _atomic_write(path: Path, text: str) -> None:
    tmp = path.with_suffix(path.suffix + ".tmp")
    tmp.write_text(text, encoding="utf-8")
    os.replace(tmp, path)


def _dump(path: Path, payload: dict) -> None:
    _atomic_write(path, json.dumps(payload, indent=2, ensure_ascii=False) + "\n")


def patch_dq_layout() -> None:
    dashboard = json.loads(DQ.read_text(encoding="utf-8"))
    for panel in dashboard.get("panels") or []:
        gp = panel.get("gridPos") or {}
        y = int(gp.get("y") or 0)
        pid = panel.get("id")
        if y >= 19 and pid not in {9404, 9406, 9405, 9101, 9102, 9103, 9400, 9401, 1000}:
            gp = dict(gp)
            gp["y"] = y + 1
            panel["gridPos"] = gp
        if pid != 9406:
            continue
        for override in panel.get("fieldConfig", {}).get("overrides") or []:
            field = str((override.get("matcher") or {}).get("options") or "")
            if field not in {"status", "Started", "Completed", "Run", "run_id"}:
                continue
            props = list(override.get("properties") or [])
            if any(item.get("id") == "custom.hidden" for item in props):
                continue
            props.append({"id": "custom.hidden", "value": True})
            override["properties"] = props
    _dump(DQ, dashboard)
